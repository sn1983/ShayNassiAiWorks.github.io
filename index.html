(function(){ const owner = 'sn1983'; const repo = 'ShayNassiAiWorks.github.io'; const branch = 'main'; // שימוש ב-jsDelivr במקום ב-GitHub Raw לצורך הזרמת וידאו תקינה function getFileUrl(path) { return `https://cdn.jsdelivr.net/gh/${owner}/${repo}@${branch}/${encodeURI(path)}`; } const imgRE = /\.(png|jpe?g|gif|webp|svg)$/i; const vidRE = /\.(mp4|webm|ogg|mov)$/i; async function listContents(path = ''){ const api = `https://api.github.com/repos/${owner}/${repo}/contents/${path}?ref=${branch}`; const res = await fetch(api); if(!res.ok) throw new Error('API Error: ' + res.status); return res.json(); } async function gatherFiles(path = ''){ const items = await listContents(path); let files = []; const list = Array.isArray(items) ? items : [items]; for(const it of list){ if(it.type === 'file'){ const isImg = imgRE.test(it.name); const isVid = vidRE.test(it.name); if(isImg || isVid){ files.push({ name: it.path, url: getFileUrl(it.path), // הכתובת החדשה type: isImg ? 'image' : 'video' }); } } else if(it.type === 'dir'){ files = files.concat(await gatherFiles(it.path)); } } return files; } // שאר הפונקציות (makeItem, openModal, closeModal) נשארות כפי שהיו בגרסה הקודמת // חשוב לוודא שב-makeItem עבור וידאו יש: // media.preload = "metadata"; })();
